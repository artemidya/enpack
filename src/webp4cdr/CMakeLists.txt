cmake_minimum_required(VERSION 3.20)
project(WEBP4CDR27 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
foreach(option ANIM_UTILS CWEBP DWEBP GIF2WEBP IMG2WEBP VWEBP WEBPINFO LIBWEBPMUX WEBPMUX EXTRAS)
 set(WEBP_BUILD_${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(WEBP_USE_THREAD OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
if(NOT WEBP_SOURCE)
 include(FetchContent)
 FetchContent_Declare(webpsource URL https://codeload.github.com/webmproject/libwebp/tar.gz/4fa21912338357f89e4fd51cf2368325b59e9bd9 URL_HASH SHA256=923f3382a47a2af185c3240c954cf004428b237bd7317413a95146d01eb4b94b)
 FetchContent_MakeAvailable(webpsource)
 set(WEBP_SOURCE ${webpsource_SOURCE_DIR})
else()
 add_subdirectory(${WEBP_SOURCE} libwebp)
endif()
add_library(WEBP4CDRx64 SHARED webp4cdr.cpp)
set_target_properties(WEBP4CDRx64 PROPERTIES PREFIX "" SUFFIX ".cpg")
target_include_directories(WEBP4CDRx64 PRIVATE ${WEBP_SOURCE}/src)
target_compile_definitions(WEBP4CDRx64 PRIVATE NOMINMAX _WIN32_WINNT=0x0601)
target_link_libraries(WEBP4CDRx64 PRIVATE webpdecoder ole32 oleaut32 shell32 user32 shlwapi windowscodecs uuid)
if(MINGW)
 target_link_options(WEBP4CDRx64 PRIVATE -static -s)
endif()
option(ADDON_TESTS "Build Windows test harness" OFF)
if(ADDON_TESTS)
 add_executable(webp27_tests ../../tools/test_webp_windows.cpp)
 target_include_directories(webp27_tests PRIVATE ${WEBP_SOURCE}/src)
 target_compile_definitions(webp27_tests PRIVATE NOMINMAX _WIN32_WINNT=0x0601)
 target_link_libraries(webp27_tests PRIVATE webpdecoder ole32 oleaut32 shell32 user32 shlwapi windowscodecs uuid)
endif()
